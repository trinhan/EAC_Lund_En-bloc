4.3 Correlated clinical variables

Using kendall Tau’s rank based test to assess associations between (ordinal) variables, we find that:

We find that:

  • Stage is strongly correlated with tumour size and node status
  • Node status is associated with shorter follow-up time

Red indicates positive association, blue indicates negative association. Statistically significant associations following BH multiple hypothesis testing are asterisked (** p < 0.05; *p<0.1)

index <- c("Stage", "Grade", "pT", "pN", "pM", "LN.Ratio", "Age.at.op", "Follow.up.Survival..months.", "Gender.Male.1..F.0", "Location")

Stemp <- S2[, index]
Stemp$Grade[which(Stemp$Grade == "Mucinous")] <- NA
DFmat <- matrix(NA, nrow = length(index), ncol = length(index))
colnames(DFmat) <- index
rownames(DFmat) <- index
DFmatp <- DFmat

for (i in 1:length(index)) {
  for (j in 1:length(index)) {
    a1 <- cor.test(as.numeric(Stemp[, i]), as.numeric(Stemp[, j]), method = "kendall", use = "complete")
    DFmat[i, j] <- a1$statistic
    DFmatp[i, j] <- a1$p.value
  }
}

diag(DFmatp) <- 1
DFmatp2 <- matrix(p.adjust(DFmatp), nrow = length(index)) / 2

dfnew <- melt(DFmat)
dfm2 <- melt(DFmatp2)
dfnew$p <- dfm2$value
dfnew$sig <- ifelse(dfnew$p < 0.1, "*", ifelse(dfnew$p < 0.05, "**", ""))

ggplot(data = dfnew, aes(x = Var1, y = Var2, fill = value)) +
  geom_tile() +
  scale_fill_gradient2(low = "#2C7BB6", mid = "white", high = "#D7191C") +
  geom_text(aes(label = sig), color = "black", size = 5) +
  labs(y = NULL, x = NULL, fill = "tau") +
  theme_bw() +
  theme(axis.text.x = element_text(angle = -45, hjust = 0))

Perform a univariate cox regression model for survival status for the above variables, and list in a forest plot.

Notes:

  • LNR/Node status and AGE is associated with survival status
  • Stage association with survival status: mainly for stage 3 samples
  • Location is significant predictor ONLY when age is taken into account
S2$Location <- S2$Cancer.Location.By.Pathologist

## Update November 2025: replace existing Alive variable with Censor Status

ParamID <- c("Age.at.op", "Gender.Male.1..F.0", "Grade", "Stage", "pT", "pN", "pM", "Cancer.Location.By.Pathologist", "LN.Ratio")
# colnames(S2)[c(9:11, 13:18, 20)]
nx <- sapply(ParamID, function(x) levels((S2[, x])))
nl <- sapply(nx, length)
nx[which(nl == 0)] <- ParamID[which(nl == 0)]
nl <- sapply(nx, length)
allparam <- data.frame(Param = character(), Coeff = numeric(), Lower = numeric(), Upper = numeric(), P = numeric(), Psum = numeric())

for (i in 1:length(ParamID)) {
  ax <- coxph(ss ~ (S2[, ParamID[i]]), data = S2, id = S2$Name)
  lx <- coefficients(summary(ax))
  lx2 <- (summary(ax))$conf.int
  summary_ax <- summary(ax)
  coef_table <- summary_ax$coefficients
  confint_table <- summary_ax$conf.int
  Out <- data.frame(
    Var = rownames(coef_table),
    Coeff = coef_table[, "exp(coef)"],
    Lower = confint_table[, "lower .95"],
    Upper = confint_table[, "upper .95"],
    P = coef_table[, "Pr(>|z|)"],
    Extra = NA
  )
  colnames(Out) <- colnames(allparam)
  if (nl[i] > 1) {
    Out <- rbind(rep(0, ncol(Out)), Out)
  }
  allparam <- rbind(allparam, Out)
}

if (any(Out$P < 0, na.rm = TRUE)) {

4.6 Clinical correlates with survival status (Fig1C)

Construct a multivariate cox regression model to identify clinical predictors of survival status, including age, stage, grade, location, Gender and Lymph Node Ratio

Negative predictors of survival status include:

  • Age at operation (note log scaled to make roughly normal)
  • Lymph node ratio: ratio of the number of positive lymph nodes to the total number of dissected lymph nodes. Note that this has been reported to be associated with patient survival status, and our clinical cohort shows this.
  • Distal esofaguscancer associated with poorer survival status compared to Cardia

Note that the median age is 72

ss <- Surv(S2$Follow.up.Survival..months., S2$CensorStatus)
## +S2$LNR+S2$pT+S2$pN+S2$pT+S2$Grade
a1 <- coxph(ss ~ log(S2$Age.at.op) + S2$Stage + S2$Location + S2$Gender.Male.1..F.0)
ax <- summary(a1)

## create a forest plot here
ParamID <- c("Age.at.op", "Stage", "Location", "Gender.Male.1..F.0")
nx <- sapply(ParamID, function(x) levels((S2[, x])))
nl <- sapply(nx, length)
nx[which(nl == 0)] <- ParamID[which(nl == 0)]
nl <- sapply(nx, length)
allparam <- data.frame(Param = character(), Coeff = numeric(), Lower = numeric(), Upper = numeric(), P = numeric(), Psum = numeric())

for (i in 1:length(ParamID)) {
  idx2 <- grep(ParamID[i], rownames(ax$coefficients))
  Out <- cbind(
    rownames(ax$coefficients)[idx2], ax$coefficients[idx2, 2],
    ax$conf.int[idx2, 3], ax$conf.int[idx2, 4], ax$coefficients[idx2, 5], ax$logtest[3]
  )
  colnames(Out) <- colnames(allparam)
  if (nl[i] > 1) {
    Out <- rbind(rep(NA, ncol(Out)), Out)
  }
  allparam <- rbind(allparam, Out)
}

allparam$Param <- unlist(nx)
allparam$Field <- rep(ParamID, times = nl)
allparam$Coeff <- round(as.numeric(allparam$Coeff), 2)
allparam$Lower <- as.numeric(allparam$Lower)
allparam$Upper <- as.numeric(allparam$Upper)
allparam$P <- round(as.numeric(allparam$P), 2)
forestplot(allparam[, c(7, 1, 2, 5)], allparam$Coeff, allparam$Lower, allparam$Upper, xlog = T, boxsize = 0.1)
