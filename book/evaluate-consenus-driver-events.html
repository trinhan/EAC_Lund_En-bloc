We identify specific mutational Driver Events as:

  • indicated in Oncokb (oncogenic or likely oncogenic) OR CGI oncogenic Class (predicted or known) or CMC non-0 score
  • CGI oncogenic Class (non-protein affecting) AND Consequence 5’UTR AND OncodriveFML promoter as predicted Promoter affecting

The type of variant (activating or LoF) is indicated by:

  • Oncokb or CGI role. If neither is available, use CMC
  • Potential Drivers which do not have a predicted outcome are indicated as unknonw
## List whether a variant is a driver
FinalSomaticList$DriverStatus <- ifelse(FinalSomaticList$CGI_oncogenic_class %in% c("known driver", "predicted") | FinalSomaticList$CMC > 0 | FinalSomaticList$ONCOKB_class %in% c("Likely Oncogenic", "Oncogenic"), "Driver", NA)

# OLD
# FinalSomaticList$DriverStatus=ifelse(FinalSomaticList$CGI_oncogenic_class%in%c("known driver", "predicted")|FinalSomaticList$CGI_role%in%c("Act", "LoF")|FinalSomaticList$ONCOKB_class%in%c("Likely Oncogenic", "Oncogenic"), "Driver", NA)

FinalSomaticList <- FinalSomaticList %>%
  mutate(DriverEffect = case_when(
    # Specific annotations take priority
    CGI_role == "LoF" | ONCOKB_effect %in% c("Loss-of-function", "Likely Loss-of-function") ~ "LoF",
    CGI_role == "Act" | ONCOKB_effect %in% c("Gain-of-function", "Likely Gain-of-function") ~ "GoF",

    # Gene-level defaults as fallback
    TSG_OCG == "tsg" ~ "LoF",
    TSG_OCG == "onc" ~ "GoF",

    # Generic driver status as last resort
    DriverStatus == "Driver" ~ "Unknown",

    # Default for non-drivers
    TRUE ~ NA_character_
  ))

Create a new table specifically for driver mutations, and condense them such that we can easily visualise them. This section also uses CNV changes (output from Oncokb)

# Filter SNVs based on the Somatic List
Driver1 <- FinalSomaticList %>%
  filter(DriverStatus == "Driver") %>%
  select(L1 = Tumor_Sample_Barcode, Gene = Hugo_Symbol, Mut = HGVSp_Short, loh, gene_loh, Driver = DriverStatus, Cons = DriverEffect) %>%
  mutate(Vtype = "SNV", value = paste(Gene, Mut), gene_loh = as.character(gene_loh))

# Extract CNV information from Oncokb
Driver2 <- strsplit(OncokbDFs$ONCOGENIC_MUTATIONS, ";")
names(Driver2) <- OncokbDFs$SAMPLE_ID
DriverCNV <- melt(Driver2) %>%
  mutate(
    Vtype = case_when(
      grepl("Del", value) ~ "DEL",
      grepl("Amp", value) ~ "AMP",
      TRUE ~ "SNV"
    ),
    Gene = sub(" .*", "", value),
    Mut = Vtype,
    Driver = "CNV",
    loh = "",
    gene_loh = "",
    Cons = Vtype
  ) %>%
  filter(Vtype %in% c("DEL", "AMP"))

# Combine SNV and CNV events
DriverEvents <- bind_rows(Driver1, DriverCNV) %>%
  mutate(Alive_edited = SampleIDs$Alive_edited[match(L1, SampleIDs$SAMPLE_ID)])

## Consolidate information for genes that have both SNV and CNV in the same sample
DriverEvents <- DriverEvents %>%
  group_by(L1, Gene) %>%
  mutate(
    any_cnv = any(Vtype %in% c("AMP", "DEL")),
    cnv_vtypes = paste(unique(Vtype[Vtype %in% c("AMP", "DEL")]), collapse = " "),
    cnv_cons = paste(unique(Cons[Vtype %in% c("AMP", "DEL")]), collapse = " "),
    Vtype = if_else(Vtype == "SNV" & any_cnv, paste(Vtype, cnv_vtypes), Vtype),
    Cons = if_else(Vtype == "SNV" & any_cnv, paste(Cons, cnv_cons), Cons)
  ) %>%
  ungroup() %>%
  select(-any_cnv, -cnv_vtypes, -cnv_cons)
